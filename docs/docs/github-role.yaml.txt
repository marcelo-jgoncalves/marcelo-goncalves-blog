AWSTemplateFormatVersion: '2010-09-09'
Description: >
  (Execute PARA CADA REPO)
  Cria uma IAM Role para um repositório GitHub específico assumir via OIDC.

Parameters:
  # --- Parâmetros do GitHub ---
  GitHubOrg:
    Type: String
    Description: "Seu nome de usuário ou organização do GitHub (ex: meu-usuario-github)"

  GitHubRepo:
    Type: String
    Description: "O nome do seu repositório no GitHub (ex: meu-projeto-legal)"

  GitHubBranch:
    Type: String
    Default: main
    Description: "O nome da branch que pode assumir esta role (ex: main, develop)"

  # --- Parâmetros da AWS ---
  RoleName:
    Type: String
    Description: "Um nome único para a IAM Role (ex: GitHub-MeuProjeto-Role)"

  OidcProviderArn:
    Type: String
    Description: O ARN do Provedor OIDC do GitHub (obtido do Template 1).
    MinLength: 20

  ManagedPolicyArn:
    Type: String
    Default: 'arn:aws:iam::aws:policy/AmazonS3ReadOnlyAccess'
    Description: "O ARN da política de permissões que esta role terá (ex: S3, EC2)"

Resources:
  GitHubRepoRole:
    Type: AWS::IAM::Role
    Properties:
      RoleName: !Ref RoleName
      ManagedPolicyArns:
        - !Ref ManagedPolicyArn
      AssumeRolePolicyDocument:
        Version: '2012-10-17'
        Statement:
          - Effect: Allow
            Principal:
              Federated: !Ref OidcProviderArn
            Action: sts:AssumeRoleWithWebIdentity
            Condition:
              # Restringe a role APENAS para o repositório e branch especificados
              StringLike:
                'token.actions.githubusercontent.com:sub': !Sub 'repo:${GitHubOrg}/${GitHubRepo}:ref:refs/heads/${GitHubBranch}'
              # Garante que o público (audience) é a AWS
              StringEquals:
                'token.actions.githubusercontent.com:aud': 'sts.amazonaws.com'

Outputs:
  RoleArn:
    Description: O ARN da Role que você deve usar no seu workflow do GitHub.
    Value: !GetAtt GitHubRepoRole.Arn
